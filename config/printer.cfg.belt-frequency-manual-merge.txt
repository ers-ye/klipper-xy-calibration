# B2K belt frequency capture — manual merge snippet only.
# Copy these sections into the existing printer.cfg; do not include this file.
# The local Node collector must be running before BELT_X or BELT_Y is used.
# These macros only emit marker responses and never move the printer.

[gcode_macro BELT_X]
description: Start passive acceleration capture for belt X
gcode:
  RESPOND TYPE=echo MSG="B2K_BELT_SESSION_START_X"

[gcode_macro BELT_Y]
description: Start passive acceleration capture for belt Y
gcode:
  RESPOND TYPE=echo MSG="B2K_BELT_SESSION_START_Y"

[gcode_macro OK]
description: Finish the active B2K belt frequency capture
gcode:
  RESPOND TYPE=echo MSG="B2K_BELT_SESSION_FINISH"
